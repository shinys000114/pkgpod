In a GitHub Actions workflow 
permissions:
  id-token: write   # the job may ask GitHub for its OIDC token
  contents: read
steps:
  - uses: actions/checkout@v4
  - uses: <owner>/pkgpod@master          # the pkgpod action
    with:
      url: <this server>
      repository: ppa
      files: dist/*.deb
      params: dists=noble
# or by hand:
  - run: |
      curl -fsSL <this server>/cli/install.sh | sh
      pkgpod --url <this server> login --oidc
      pkgpod upload ppa dist/*.deb -s dists=noble
